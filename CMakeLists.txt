cmake_minimum_required(VERSION 3.20)
project(WinAIOOptimize VERSION 6.0.1 LANGUAGES C RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows 10/11 and a Windows C compiler are required")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(win_aio_optimize WIN32 src/app.c src/network.c src/backup.c src/optimizer.c src/packages.c src/protection.c src/theme.c src/system_info.c src/app.rc)
target_include_directories(win_aio_optimize PRIVATE src)
target_compile_definitions(win_aio_optimize PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(win_aio_optimize PRIVATE
  user32 gdi32 comctl32 comdlg32 shell32 shlwapi ole32 uuid uxtheme
  advapi32 iphlpapi ws2_32 wlanapi powrprof runtimeobject dwmapi oleaut32 wbemuuid dxgi dxguid)
if(MSVC)
  target_compile_options(win_aio_optimize PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/utf-8>")
  target_link_options(win_aio_optimize PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
else()
  target_compile_options(win_aio_optimize PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror>")
  target_link_options(win_aio_optimize PRIVATE -municode -Wl,--dynamicbase,--nxcompat)
endif()
set_target_properties(win_aio_optimize PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
add_custom_command(TARGET win_aio_optimize POST_BUILD
  COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -NonInteractive -ExecutionPolicy Bypass
    -File "${CMAKE_SOURCE_DIR}/scripts/write-checksum.ps1" -Binary "$<TARGET_FILE:win_aio_optimize>"
  VERBATIM)
enable_testing()
add_test(NAME read_only_self_test COMMAND win_aio_optimize --self-test)
add_test(NAME read_only_ui_english COMMAND win_aio_optimize --smoke-test --lang=en --theme=light)
add_test(NAME read_only_ui_vietnamese COMMAND win_aio_optimize --smoke-test --lang=vi --theme=dark)
add_test(NAME read_only_ui_english_dark COMMAND win_aio_optimize --smoke-test --lang=en --theme=dark)
add_test(NAME read_only_ui_vietnamese_light COMMAND win_aio_optimize --smoke-test --lang=vi --theme=light)
set_tests_properties(read_only_self_test read_only_ui_english read_only_ui_vietnamese read_only_ui_english_dark read_only_ui_vietnamese_light PROPERTIES TIMEOUT 45)
